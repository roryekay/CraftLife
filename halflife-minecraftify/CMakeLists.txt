cmake_minimum_required(VERSION 3.20)
project(MinecraftifyAnything LANGUAGES CXX)
include(CTest)
add_library(fusion_protocol INTERFACE)
target_include_directories(fusion_protocol INTERFACE native/include)
target_compile_features(fusion_protocol INTERFACE cxx_std_17)
add_executable(frame_protocol_test tests/frame_protocol.cpp)
target_link_libraries(frame_protocol_test PRIVATE fusion_protocol)
add_test(NAME frame_protocol COMMAND frame_protocol_test)
add_executable(valve_wire_test tests/valve_wire.cpp)
target_compile_features(valve_wire_test PRIVATE cxx_std_17)
target_compile_definitions(valve_wire_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME valve_wire COMMAND valve_wire_test)
add_executable(movement_test tests/movement.cpp)
target_compile_features(movement_test PRIVATE cxx_std_17)
add_test(NAME movement_sweeps COMMAND movement_test)
add_executable(reprojection_test tests/reprojection.cpp)
target_link_libraries(reprojection_test PRIVATE fusion_protocol)
add_test(NAME camera_reprojection COMMAND reprojection_test)
if(WIN32)

  target_compile_definitions(MinecraftifyRenderer PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    set_target_properties(MinecraftifyRenderer PROPERTIES SUFFIX ".addon32")
  else()
    set_target_properties(MinecraftifyRenderer PROPERTIES SUFFIX ".addon64")
  endif()

endif()
