cmake_minimum_required(VERSION 3.20)
project(MinecraftifyHL2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(PORT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SDK "${PORT}/local/sdk/src")
add_library(MinecraftifyHL2 SHARED plugin.cpp)
target_include_directories(MinecraftifyHL2 PRIVATE "${SDK}/public" "${SDK}/public/tier0" "${SDK}/public/tier1" "${SDK}/common" "${SDK}/game/shared" "${PORT}/../halflife-minecraftify/native/include")
target_compile_definitions(MinecraftifyHL2 PRIVATE WIN32 _WINDOWS NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(MinecraftifyHL2 PRIVATE /W0 /MP4)
target_link_libraries(MinecraftifyHL2 PRIVATE ws2_32 "${SDK}/lib/public/tier0.lib" "${SDK}/lib/public/vstdlib.lib")
