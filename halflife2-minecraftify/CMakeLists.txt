cmake_minimum_required(VERSION 3.20)
project(MinecraftifyAnything LANGUAGES CXX)
include(CTest)
add_library(fusion_protocol INTERFACE)
target_include_directories(fusion_protocol INTERFACE native/include)
target_compile_features(fusion_protocol INTERFACE cxx_std_17)
add_executable(frame_protocol_test tests/frame_protocol.cpp)
target_link_libraries(frame_protocol_test PRIVATE fusion_protocol)
add_test(NAME frame_protocol COMMAND frame_protocol_test)
if(WIN32)
  add_library(MinecraftifyRenderer SHARED native/src/renderer.cpp)
  target_link_libraries(MinecraftifyRenderer PRIVATE fusion_protocol)
  target_include_directories(MinecraftifyRenderer PRIVATE third_party/reshade)
  target_compile_definitions(MinecraftifyRenderer PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  set_target_properties(MinecraftifyRenderer PROPERTIES SUFFIX ".addon64")
  add_executable(frame_probe native/src/frame_probe.cpp)
  target_link_libraries(frame_probe PRIVATE fusion_protocol)
  target_compile_definitions(frame_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
